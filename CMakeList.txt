cmake_minimum_required(VERSION 3.16)

project(Projet_Geometrie
    VERSION 1.0
    DESCRIPTION "Application console de geometrie en C++ orientee objet"
    LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(Projet_Geometrie
    main.cpp
    src/Point.cpp
    src/Polygone.cpp
    src/Cercle.cpp
    src/Triangle.cpp
    src/Rectangle.cpp
    src/Carre.cpp
)

target_include_directories(Projet_Geometrie PRIVATE include)

if (MSVC)
    target_compile_options(Projet_Geometrie PRIVATE /W4)
else()
    target_compile_options(Projet_Geometrie PRIVATE -Wall -Wextra -Wpedantic)
endif()
